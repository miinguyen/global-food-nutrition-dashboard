% ═══════════ SLIDE 3 — DATASET DESCRIPTION ═════════════════
\begin{frame}
\begin{center}
  {\large\bfseries\textcolor{NDarkGreen}{Dataset Description}}
\end{center}
\vspace{-2pt}
\begin{columns}[T]
  \begin{column}{0.32\textwidth}
    \begin{tcolorbox}[ybox, title={\footnotesize\bfseries Q1: What does the world eat?},
      coltitle=NBrown, colbacktitle=NYellow!35]
      \footnotesize
      \textbf{Data:} OWID --- Calorie Supply \& Food Composition\\[1pt]
      \textbf{Scope:} 200+ countries, 1961--2023\\[1pt]
      Per-capita calories by food group.
    \end{tcolorbox}
  \end{column}
  \begin{column}{0.32\textwidth}
    \begin{tcolorbox}[gbox, title={\footnotesize\bfseries Q2: How healthy is it?},
      coltitle=white, colbacktitle=NGreen]
      \footnotesize
      \textbf{Data:} OWID Obesity \& Life Exp.; WHO GHO\\[1pt]
      \textbf{Scope:} 200+ countries, 1975--2023\\[1pt]
      Obesity, life expectancy, child underweight.
    \end{tcolorbox}
  \end{column}
  \begin{column}{0.32\textwidth}
    \begin{tcolorbox}[obox, title={\footnotesize\bfseries Q3: What's in our food?},
      coltitle=white, colbacktitle=NOrange]
      \footnotesize
      \textbf{Data:} Open Food Facts\\[1pt]
      \textbf{Scope:} $\sim$10\,000 products\\[1pt]
      Nutri-Score (A--E), NOVA (1--4), macros.
    \end{tcolorbox}
  \end{column}
\end{columns}
\vspace{3pt}
\begin{tcolorbox}[dbox]
  \footnotesize
  \textbf{Integration:} All country-level sources linked via \textbf{ISO-3166 codes}
  with automated download scripts.
  Key challenge: reconciling 60-year national aggregates with product-level granularity.
\end{tcolorbox}
\end{frame}
